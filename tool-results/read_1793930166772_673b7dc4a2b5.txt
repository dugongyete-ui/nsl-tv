    46→// ============================================================
    47→// Util
    48→// ============================================================
    49→function b64urlEncode(str: string): string {
    50→  const bytes = new TextEncoder().encode(str);
    51→  let bin = '';
    52→  bytes.forEach((b) => (bin += String.fromCharCode(b)));
    53→  return btoa(bin).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    54→}
    55→function hh(src: TSource): string {
    56→  return b64urlEncode(JSON.stringify(src.headers ?? {}));
    57→}
    58→function proxied(src: TSource): string {
    59→  return `/api/stream?u=${encodeURIComponent(src.url).replace(/%24/g, '$')}&hh=${hh(src)}`;
    60→}
    61→function proxiedLicense(src: TSource, licUrl: string): string {
    62→  return `/api/license?u=${encodeURIComponent(licUrl)}&hh=${hh(src)}`;
    63→}
    64→function hexToB64url(hex: string): string {
    65→  const clean = hex.replace(/[^0-9a-fA-F]/g, '');
    66→  const bytes = new Uint8Array(clean.length / 2);
    67→  for (let i = 0; i < bytes.length; i++) bytes[i] = parseInt(clean.substr(i * 2, 2), 16);
    68→  let bin = '';
    69→  bytes.forEach((b) => (bin += String.fromCharCode(b)));
    70→  return btoa(bin).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    71→}
    72→function now(): string {
    73→  return new Date().toLocaleTimeString('id-ID', { hour12: false });
    74→}
    75→function typeOf(src: TSource): string {
    76→  if (src.manifestType) return src.manifestType.toUpperCase();
    77→  const u = (src.url || '').split('?')[0].toLowerCase();
    78→  if (u.endsWith('.mpd')) return 'MPD';
    79→  if (u.endsWith('.m3u8')) return 'HLS';
    80→  return 'HLS';
    81→}
