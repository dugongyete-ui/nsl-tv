   115→   115→    private void boot() {
   116→   116→        booted.set(false);
   117→   117→        new Thread(() -> {
   118→   118→            try {
   119→   119→                // telemetri (tanpa menunggu)
   120→   120→                new Thread(() -> {
   121→   121→                    try { Api.postPlain("/api/v1/devices/ping", Store.deviceId(MainActivity.this)); }
   122→   122→                    catch (Exception ignored) {}
   123→   123→                }).start();
   124→   124→
   125→   125→                // 1. lisensi
   126→   126→                setBoot(getString(R.string.boot_license));
   127→   127→                Models.License lic = fetchLicense();
   128→   128→                license = lic;
   129→   129→                if (!lic.allows()) {
   130→   130→                    ui.post(() -> showActivation());
   131→   131→                    return;
   132→   132→                }
   133→   133→
   134→   134→                // 2. saluran
   135→   135→                setBoot(getString(R.string.boot_channels));
   136→   136→                Data.parseChannels(Api.get("/api/v1/channels", Store.deviceId(this)));
   137→   137→
   138→   138→                setBoot(getString(R.string.boot_ready));
   139→   139→                ui.postDelayed(this::enterMain, 350);
   140→   140→            } catch (final Api.HttpException he) {
   141→   141→                if (he.body != null && he.body.contains("diblokir")) {
   142→   142→                    ui.post(() -> {
   143→   143→                        blockedMsg.setText("Perangkat ini telah diblokir oleh administrator server.\n\nHubungi penyedia layanan untuk bantuan.");
   144→   144→                        showOnly(blockedView);
   145→   145→                    });
   146→   146→                } else {
   147→   147→                    ui.post(() -> {
   148→   148→                        blockedMsg.setText("Gagal terhubung ke server.\n" + shortMsg(he.getMessage()));
   149→   149→                        showOnly(blockedView);
   150→   150→                    });
   151→   151→                }
   152→   152→            } catch (final Exception e) {
   153→   153→                ui.post(() -> {
   154→   154→                    blockedMsg.setText("Terjadi kesalahan.\n" + shortMsg(e.getMessage()));
   155→   155→                    showOnly(blockedView);
   156→   156→                });
   157→   157→            }
   158→   158→        }, "boot").start();
   159→   159→    }

   161→   161→    private Models.License fetchLicense() throws Exception {
   162→   162→        Models.License lic = new Models.License();
   163→   163→        try {
   164→   164→            JSONObject o = new JSONObject(Api.get("/api/v1/license/status", Store.deviceId(this)));
   165→   165→            lic.enforced = o.optBoolean("enforced", false);
   166→   166→            lic.minDonation = o.optLong("minDonation", 5000L);
   167→   167→            lic.state = o.optString("state", "");
   168→   168→            lic.trialEndsAt = parseMillis(o.optString("trialEndsAt", ""));
   169→   169→        } catch (Api.HttpException he) {
   170→   170→            if (he.code == 403 && he.body != null && he.body.contains("diblokir")) throw he;
   171→   171→            lic.enforced = false; // bila endpoint lisensi bermasalah, jangan blokir
   172→   172→        }
   173→   173→        return lic;
   174→   174→    }

   409→   409→    private void wireActivation() {
   410→   410→        amountBox = findViewById(R.id.amountBox);
   411→   411→        orderCard = findViewById(R.id.orderCard);
   412→   412→        qrImage = findViewById(R.id.qrImage);
   413→   413→        orderAmount = findViewById(R.id.orderAmount);
   414→   414→        orderStatus = findViewById(R.id.orderStatus);
   415→   415→        orderExpiry = findViewById(R.id.orderExpiry);
   416→   416→        actDesc = findViewById(R.id.actDesc);
   417→   417→        btnPay = findViewById(R.id.btnPay);
   418→   418→        btnCheckPaid = findViewById(R.id.btnCheckPaid);
   419→   419→        btnCancelOrder = findViewById(R.id.btnCancelOrder);
   420→   420→
   421→   421→        btnPay.setOnClickListener(v -> {
   422→   422→            if (chosenAmount <= 0) {
   423→   423→                Toast.makeText(this, "Pilih nominal aktivasi dulu", Toast.LENGTH_SHORT).show();
   424→   424→                return;
   425→   425→            }
   426→   426→            btnPay.setEnabled(false);
   427→   427→            btnPay.setText("Membuat kode QRIS…");
   428→   428→            new Thread(() -> {
   429→   429→                try {
   430→   430→                    JSONObject body = new JSONObject();
   431→   431→                    body.put("amount", chosenAmount);
   432→   432→                    JSONObject resp = new JSONObject(Api.post("/api/v1/license/orders", Store.deviceId(this), body.toString()));
   433→   433→                    JSONObject o = resp.optJSONObject("order");
   434→   434→                    if (o == null) throw new Exception(resp.optString("message", "Gagal membuat pesanan"));
   435→   435→                    order = new Models.Order();
   436→   436→                    order.orderId = o.optString("orderId", "");
   437→   437→                    order.status = o.optString("status", "PENDING");
   438→   438→                    order.amount = o.optLong("amount", chosenAmount);
   439→   439→                    order.finalAmount = o.optLong("finalAmount", chosenAmount);
   440→   440→                    order.qrImage = o.optString("qrImage", "");
   441→   441→                    order.expiresAt = parseMillis(o.optString("expiresAt", ""));

   832→   832→    private static long parseMillis(String iso) {
   833→   833→        if (iso == null || iso.isEmpty()) return 0L;
   834→   834→        try {
   835→   835→            if (iso.length() >= 10) {
   836→   836→                SimpleDateFormat isoF = new SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss", Locale.US);
   837→   837→                java.util.Date d = isoF.parse(iso.substring(0, Math.min(19, iso.length())));
   838→   838→                if (d != null) return d.getTime();
   839→   839→            }
   840→   840→            return Long.parseLong(iso);
   841→   841→        } catch (Exception e) {
   842→   842→            return 0L;
   843→   843→        }
   844→   844→    }
