   115→    private void boot() {
   116→        booted.set(false);
   117→        new Thread(() -> {
   118→            try {
   119→                // telemetri (tanpa menunggu)
   120→                new Thread(() -> {
   121→                    try { Api.postPlain("/api/v1/devices/ping", Store.deviceId(MainActivity.this)); }
   122→                    catch (Exception ignored) {}
   123→                }).start();
   124→
   125→                // 1. lisensi
   126→                setBoot(getString(R.string.boot_license));
   127→                Models.License lic = fetchLicense();
   128→                license = lic;
   129→                if (!lic.allows()) {
   130→                    ui.post(() -> showActivation());
   131→                    return;
   132→                }
   133→
   134→                // 2. saluran
   135→                setBoot(getString(R.string.boot_channels));
   136→                Data.parseChannels(Api.get("/api/v1/channels", Store.deviceId(this)));
   137→
   138→                setBoot(getString(R.string.boot_ready));
   139→                ui.postDelayed(this::enterMain, 350);
   140→            } catch (final Api.HttpException he) {
   141→                if (he.body != null && he.body.contains("diblokir")) {
   142→                    ui.post(() -> {
   143→                        blockedMsg.setText("Perangkat ini telah diblokir oleh administrator server.\n\nHubungi penyedia layanan untuk bantuan.");
   144→                        showOnly(blockedView);
   145→                    });
   146→                } else {
   147→                    ui.post(() -> {
   148→                        blockedMsg.setText("Gagal terhubung ke server.\n" + shortMsg(he.getMessage()));
   149→                        showOnly(blockedView);
   150→                    });
   151→                }
   152→            } catch (final Exception e) {
   153→                ui.post(() -> {
   154→                    blockedMsg.setText("Terjadi kesalahan.\n" + shortMsg(e.getMessage()));
   155→                    showOnly(blockedView);
   156→                });
   157→            }
   158→        }, "boot").start();
   159→    }

   161→    private Models.License fetchLicense() throws Exception {
   162→        Models.License lic = new Models.License();
   163→        try {
   164→            JSONObject o = new JSONObject(Api.get("/api/v1/license/status", Store.deviceId(this)));
   165→            lic.enforced = o.optBoolean("enforced", false);
   166→            lic.minDonation = o.optLong("minDonation", 5000L);
   167→            lic.state = o.optString("state", "");
   168→            lic.trialEndsAt = parseMillis(o.optString("trialEndsAt", ""));
   169→        } catch (Api.HttpException he) {
   170→            if (he.code == 403 && he.body != null && he.body.contains("diblokir")) throw he;
   171→            lic.enforced = false; // bila endpoint lisensi bermasalah, jangan blokir
   172→        }
   173→        return lic;
   174→    }

   388→    private void updateTrialChip() {
   389→        if (license == null) return;
   390→        if (!license.enforced) {
   391→            chipTrial.setVisibility(View.GONE);
   392→            return;
   393→        }
   394→        chipTrial.setVisibility(View.VISIBLE);
   395→        if ("ACTIVE".equals(license.state)) {
   396→            chipTrial.setText("PREMIUM AKTIF");
   397→        } else if (license.trialEndsAt > 0) {
   398→            long days = (license.trialEndsAt - System.currentTimeMillis()) / 86400000L;
   399→            chipTrial.setText(String.format(Locale.US, "TRIAL • %d HARI", Math.max(0, days)));
   400→        } else {
   401→            chipTrial.setText("TRIAL");
   402→        }
   403→    }

   422→            if (chosenAmount <= 0) {
   423→                Toast.makeText(this, "Pilih nominal aktivasi dulu", Toast.LENGTH_SHORT).show();
   424→                return;
   425→            }
   426→            btnPay.setEnabled(false);
   427→            btnPay.setText("Membuat kode QRIS…");
   428→            new Thread(() -> {
   429→                try {
   430→                    JSONObject body = new JSONObject();
   431→                    body.put("amount", chosenAmount);
   432→                    JSONObject resp = new JSONObject(Api.post("/api/v1/license/orders", Store.deviceId(this), body.toString()));
   433→                    JSONObject o = resp.optJSONObject("order");
   434→                    if (o == null) throw new Exception(resp.optString("message", "Gagal membuat pesanan"));
   435→                    order = new Models.Order();
   436→                    order.orderId = o.optString("orderId", "");
   437→                    order.status = o.optString("status", "PENDING");
   438→                    order.amount = o.optLong("amount", chosenAmount);
   439→                    order.finalAmount = o.optLong("finalAmount", chosenAmount);
   440→                    order.qrImage = o.optString("qrImage", "");
   441→                    order.expiresAt = parseMillis(o.optString("expiresAt", ""));
   442→                    ui.post(this::showOrder);
   443→                } catch (final Exception e) {
   444→                    ui.post(() -> {
   445→                        btnPay.setEnabled(true);
   446→                        btnPay.setText("Lanjut Pembayaran QRIS");
   447→                        Toast.makeText(this, shortMsg(e.getMessage()), Toast.LENGTH_LONG).show();
   448→                    });
   449→                }
   450→            }, "order").start();
   451→        });
   452→
   453→        btnCheckPaid.setOnClickListener(v -> pollOrderOnce());
   454→        btnCancelOrder.setOnClickListener(v -> cancelOrder());
   455→    }

   457→    private void showActivation() {

   464→        long[] presets = {min, min * 2, min * 4, min * 10};
   465→        amountBox.removeAllViews();
   466→        LinearLayout row = null;

   468→            if (i % 2 == 0) {

   479→            chip.setText(rupiah(amount));
   480→            chip.setTextSize(14f);

   512→    private void showOrder() {
   513→        btnPay.setEnabled(true);
   514→        btnPay.setVisibility(View.GONE);
   515→        btnCheckPaid.setVisibility(View.VISIBLE);
   516→        btnCancelOrder.setVisibility(View.VISIBLE);
   517→        orderCard.setVisibility(View.VISIBLE);
   518→        orderAmount.setText(rupiah(order.finalAmount));
   519→        orderStatus.setText("Menunggu pembayaran…");
   520→        Bitmap bmp = decodeQr(order.qrImage);

   526→    private Bitmap decodeQr(String dataUrl) {
   527→        try {
   528→            if (dataUrl == null) return null;
   529→            String b64 = dataUrl.contains(",") ? dataUrl.substring(dataUrl.indexOf(',') + 1) : dataUrl;
   530→            byte[] raw = android.util.Base64.decode(b64, android.util.Base64.DEFAULT);
   531→            return BitmapFactory.decodeByteArray(raw, 0, raw.length);
   532→        } catch (Exception e) {
   533→            return null;
   534→        }
   535→    }

   559→    private void pollOrderLoop() {
   560→        new Thread(() -> {
   561→            int tries = 0;
   562→            while (tries < 480 && order != null && activationView.getVisibility() == View.VISIBLE) {
   563→                try {
   564→                    Thread.sleep(2500);
   565→                    if (order == null || activationView.getVisibility() != View.VISIBLE) return;
   566→                    Models.Order updated = fetchOrder(order.orderId);
   567→                    if (updated != null) {
   568→                        order = updated;
   569→                        ui.post(() -> {
   570→                            if ("PAID".equalsIgnoreCase(order.status)) {
   571→                                orderStatus.setText("Pembayaran diterima ✓ Mengaktifkan…");
   572→                                stopOrderTicker();
   573→                                waitActivation();
   574→                            } else {
   575→                                orderStatus.setText("Menunggu pembayaran…");
   576→                            }
   577→                        });
   578→                        if (updated.status != null && "PAID".equalsIgnoreCase(updated.status)) return;
   579→                    }
   580→                } catch (InterruptedException ie) {
   581→                    return;
   582→                } catch (Exception ignored) {
   583→                }
   584→                tries++;
   585→            }
   586→        }, "poll-order").start();
   587→    }

   618→        new Thread(() -> {
   619→            int tries = 0;
   620→            boolean active = false;
   621→            while (tries < 15) {
   622→                try { Thread.sleep(2000); } catch (InterruptedException e) { return; }
   623→                try {
   624→                    Models.License lic = fetchLicense();
   625→                    license = lic;
   626→                    if (lic.allows()) { active = true; break; }
   627→                } catch (Exception ignored) {}
   628→                tries++;
   629→            }
   630→            final boolean ok = active;
   631→            ui.post(() -> {
   632→                stopOrderTicker();
   633→                if (ok) {
   634→                    Toast.makeText(this, "Aktivasi berhasil! Terima kasih", Toast.LENGTH_LONG).show();
   635→                    boot();
   636→                } else {
   637→                    Toast.makeText(this, "Pembayaran diproses, coba cek lagi sesaat lagi", Toast.LENGTH_LONG).show();
   638→                }
   639→            });
   640→        }, "wait-act").start();
   641→    }

   670→        Models.Order od = new Models.Order();
   671→        od.orderId = o.optString("orderId", id);
   672→        od.status = o.optString("status", "");
   673→        od.amount = o.optLong("amount", 0);

   675→        od.qrImage = o.optString("qrImage", "");

   832→    private static long parseMillis(String iso) {
   833→        if (iso == null || iso.isEmpty()) return 0L;
   834→        try {
   835→            if (iso.length() >= 10) {
   836→                SimpleDateFormat isoF = new SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss", Locale.US);
   837→                java.util.Date d = isoF.parse(iso.substring(0, Math.min(19, iso.length())));
   838→                if (d != null) return d.getTime();
   839→            }
   840→            return Long.parseLong(iso);
   841→        } catch (Exception e) {
   842→            return 0L;
   843→        }
   844→    }
